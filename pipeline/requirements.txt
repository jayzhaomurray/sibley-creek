# Macro Research Department - Python pipeline dependencies
#
# Floors chosen for Python 3.11+ (project floor; see ARCHITECTURE.md ADR-0003).
# Versions pinned with >= rather than == so security updates flow; tighten to
# == once we have a lockfile and CI matrix.

pandas>=2.2,<3              # tabular wrangling; CSV IO; resampling
python-dateutil>=2.9,<3     # robust date parsing (Valet's "d" strings, BIS time periods, etc.)

# Testing
pytest>=8.0,<9
pytest-httpx>=0.30,<0.36    # mocks httpx clients in tests; keeps tests off the live APIs
